\section{B}
\begin{equation}
H\left[\phi\right] := \int \frac{1}{2} \left( \grad \phi \right)^2 + \frac{1}{2} (V-\mu)\phi^2
\end{equation}

\begin{equation}
F\left[\phi\right] := H\left[\phi\right] + \frac{\gamma}{4}\int\phi^4 \dV
\end{equation}

\begin{equation}
F^\prime\left[\phi\right] h = H^\prime\left[\phi\right] h + \gamma \int \phi^3 h \dV
\end{equation}

\begin{equation}
F^{\prime\prime}\left[\phi\right] h g = H^{\prime\prime}\left[\phi\right] h g + 3 \gamma \int \phi^2 h g \dV
\end{equation}

\begin{equation}
\phi^n := t^n_0 \theta_0 + t^n_1 \theta^n_1
\end{equation}

\begin{equation}
F^{\prime\prime}\left[\phi\right] h g d = -F^\prime\left[\phi\right] h
\end{equation}


\begin{equation}
F\left[\phi^n\right] = t^n_0 H\left[\theta_0\right] + t^n_1 H\left[\theta^n_1\right] + \frac{\gamma}{4} \int \left(t^n_0 \theta_0 + t^n_1 \theta^n_1\right)^4 \dV
\end{equation}

\begin{equation}
F^\prime\left[\phi^n\right] h = t^n_0 H^\prime\left[\theta_0\right] h + t^n_1 H^\prime\left[\theta^n_1\right] h + \gamma \int \left(t^n_0 \theta_0 + t^n_1 \theta^n_1\right)^3 h \dV
\end{equation}

\begin{align}
\frac{d}{d t^n_0} F^\prime\left[\phi^n\right] h &= H^\prime\left[\theta_0\right] h + 3\gamma \int\left(t^n_0 \theta_0 + t^n_1 \theta^n_1\right)^2 \theta_0 h \dV  \\
\frac{d}{d t^n_1} F^\prime\left[\phi^n\right] h &= H^\prime\left[\theta^n_1\right] h + 3\gamma \int\left(t^n_0 \theta_0 + t^n_1 \theta^n_1\right)^2 \theta^n_1 h \dV
\end{align}

\begin{align}
\frac{d}{d t^n_0} F^\prime\left[\phi^n\right] h &= H^\prime\left[\theta_0\right] h + 3\gamma \int \left((t^n_0 \theta_0)^2 + 2 t^n_0 t^n_1 \theta_0 \theta^n_1 + (t^n_1 \theta^n_1)^2 \right) \theta_0 h \dV  \\
\frac{d}{d t^n_1} F^\prime\left[\phi^n\right] h &= H^\prime\left[\theta^n_1\right] h + 3\gamma \int \left((t^n_0 \theta_0)^2 + 2 t^n_0 t^n_1 \theta_0 \theta^n_1 + (t^n_1 \theta^n_1)^2 \right) \theta^n_1 h \dV
\end{align}

\begin{align}
\frac{d^2}{d t^n_0 d t^n_0} F^\prime\left[\phi^n\right] h &= 3\gamma \int \left(2 t^n_0 (\theta_0)^2 + 2 t^n_1 \theta_0 \theta^n_1 \right) \theta_0 h \dV  \\
\frac{d^2}{d t^n_1 d t^n_0 } F^\prime\left[\phi^n\right] h &= 3\gamma \int \left(2 t^n_0 \theta_0 \theta^n_1 + 2 t^n_1 (\theta^n_1)^2 \right) \theta_0 h \dV  \\
\frac{d^2}{d t^n_0 d t^n_1} F^\prime\left[\phi^n\right] h &= 3\gamma \int \left(2 t^n_0 (\theta_0)^2 + 2 t^n_0 \theta_0 \theta^n_1 \right) \theta^n_1 h \dV \\
\frac{d^1}{d t^n_1 d t^n_1} F^\prime\left[\phi^n\right] h &= 3\gamma \int \left(2 t^n_1 \theta_0 \theta^n_1 + 2 t^n_1 (\theta^n_1)^2  \right) \theta^n_1 h \dV
\end{align}
